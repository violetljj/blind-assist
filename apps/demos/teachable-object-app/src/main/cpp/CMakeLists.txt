cmake_minimum_required(VERSION 3.22.1)
project(teachable_jni)
add_library(SNPE SHARED IMPORTED)
set_target_properties(SNPE PROPERTIES IMPORTED_LOCATION "${QAIRT_ROOT}/lib/aarch64-android/libSNPE.so")
add_library(teachable_jni SHARED teachable_jni.cpp)
target_include_directories(teachable_jni PRIVATE "${QAIRT_ROOT}/include/SNPE")
target_compile_features(teachable_jni PRIVATE cxx_std_17)
target_compile_options(teachable_jni PRIVATE -O3 -fno-fast-math -ffp-contract=off)
target_link_libraries(teachable_jni SNPE log)
target_link_options(teachable_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
